\newcommand{\grp}[3]{%
  \begin{scope}[on background layer]
    \path #1 coordinate (grpA) #2 coordinate (grpB);
    \draw[draw=black!55, line width=0.7pt, fill=black!2, rounded corners=3pt] (grpA) rectangle (grpB);
  \end{scope}
  \node[anchor=north west, inner xsep=5pt, inner ysep=3pt, font=\sffamily\fontsize{8.5}{10}\selectfont\bfseries, color=accent]
    at (grpA) {#3};
}
